## 2.End-to-End Benchmarking
这里是要实现一个简单的评估脚本，逻辑概要来说，就是先warmup几轮，让GPU预热后开始计时，随机给model一个输入，进行前向传播反向传播后，用 torch.cuda.synchronize() 来实现cpu和gpu同步，最后再计算一个时间，中间时间差就是耗时。

> Problem (benchmarking_script): 4 points

```python
import torch


def benchmark(
    model: torch.nn.Module, inputs: torch.Tensor, device: torch.device | None = None
):

    if device is not None:
        model.to(device)

    model.eval()
    warmup_iters = 5
    with torch.no_grad():
        for _ in range(warmup_iters):
            model(inputs)
            if device == "cuda":
                torch.cuda.synchronize()

    import timeit

    print("Starting benchmark for forward...")
    start_time = timeit.default_timer()
    model(inputs)
    if device == "cuda":
        torch.cuda.synchronize()
    end_time = timeit.default_timer()
    elapsed_time = end_time - start_time
    print(f"Forward pass time: {elapsed_time * 1000:.2f} ms")

    print("Starting benchmark for backward...")
    start_time = timeit.default_timer()
    model(inputs).backward()
    if device == "cuda":
        torch.cuda.synchronize()
    end_time = timeit.default_timer()
    elapsed_time = end_time - start_time
    print(f"Backward pass time: {elapsed_time * 1000:.2f} ms")

    print("Benchmarking benchmark for forward and backward...")
    start_time = timeit.default_timer()
    model(inputs).backward()
    if device == "cuda":
        torch.cuda.synchronize()
    end_time = timeit.default_timer()
    elapsed_time = end_time - start_time
    print(f"Forward and backward pass time: {elapsed_time * 1000:.2f} ms")

    return


if __name__ == "__main__":
```

## 3.Optimizing Attention with FlashAttention-2
这里简单介绍一下Flash attention，它主要利用指数运算的性质，将softmax优化成online softmax，支持分块计算。

传统attention需要将完整的S和P矩阵存储在GPU全局内存中，导致巨大的内存开销。

而FlashAttention通过online softmax算法，只保存必要的统计量而不存储完整的S和P矩阵，在反向传播时基于这些统计量重新计算所需的中间结果。这是一种用计算时间换取内存空间的策略，由于GPU的高计算吞吐量，通常能获得更好的整体性能。"

> Problem (Flash Attention): 4 points

```python
import torch
from jaxtyping import Float, Bool
import einops


class FlashAttention(torch.autograd.Function):
    @staticmethod
    def forward(
        ctx,
        Q: Float[torch.Tensor, "... seq dim"],
        K: Float[torch.Tensor, "... seq dim"],
        V: Float[torch.Tensor, "... seq dim"],
        is_causal: Bool = True,
    ):
        *b, seq, dim = Q.shape
        TILE_N = 64
        scale = 1.0 / dim**0.5
        q_scale = Q * scale  # 预缩放 Q

        O = torch.zeros_like(Q)

        # L 的形状必须是 (batch_size, seq_len)
        batch_size = b[0] if len(b) > 0 else 1
        L = torch.zeros(batch_size, seq, device=Q.device, dtype=torch.float32)

        for i in range(0, seq, TILE_N):
            i_end = min(i + TILE_N, seq)
            q_chunk = q_scale[..., i:i_end, :]  # [batch..., chunk_i, dim]

            # 初始化当前 chunk 的统计量
            m_i = torch.full((*b, i_end - i), -float("inf"), device=Q.device)
            l_i = torch.zeros((*b, i_end - i), device=Q.device)
            o_i = torch.zeros_like(q_chunk)

            for j in range(0, seq if not is_causal else i + 1, TILE_N):
                j_end = min(j + TILE_N, seq)

                k_chunk = K[..., j:j_end, :]  # [batch..., chunk_j, dim]
                v_chunk = V[..., j:j_end, :]  # [batch..., chunk_j, dim]

                # 计算注意力分数 [batch..., chunk_i, chunk_j]
                S_ij = einops.einsum(q_chunk, k_chunk, "b n d, b m d -> b n m")

                # 因果掩码：只允许关注前面的位置
                if is_causal:
                    # 当前 query 块起始位置 i，key 块起始位置 j
                    # 对于 query 位置 q_pos ∈ [i, i_end)，key 位置 k_pos ∈ [j, j_end)
                    # 只允许 k_pos <= q_pos
                    rows = i_end - i
                    cols = j_end - j

                    # 创建全局位置索引
                    q_global_indices = torch.arange(
                        i, i_end, device=Q.device
                    ).unsqueeze(
                        1
                    )  # [rows, 1]
                    k_global_indices = torch.arange(
                        j, j_end, device=Q.device
                    ).unsqueeze(
                        0
                    )  # [1, cols]

                    # 因果掩码：k_global_indices <= q_global_indices
                    causal_mask = k_global_indices <= q_global_indices
                    S_ij.masked_fill_(~causal_mask, -float("inf"))

                # 数值稳定的 softmax 计算
                m_ij = torch.max(S_ij, dim=-1, keepdim=True)[
                    0
                ]  # [batch..., chunk_i, 1]
                p_ij = torch.exp(S_ij - m_ij)  # [batch..., chunk_i, chunk_j]
                l_ij = torch.sum(p_ij, dim=-1)  # [batch..., chunk_i]

                # 在线 softmax 更新
                m_new = torch.maximum(m_i, m_ij.squeeze(-1))  # [batch..., chunk_i]

                # 数值稳定的归一化系数更新
                alpha = torch.exp(m_i - m_new)  # [batch..., chunk_i]
                beta = torch.exp(m_ij.squeeze(-1) - m_new)  # [batch..., chunk_i]

                l_new = alpha * l_i + beta * l_ij  # [batch..., chunk_i]

                # 更新输出
                P_ij_v = einops.einsum(p_ij, v_chunk, "b n m, b m d -> b n d")

                o_i = (
                    alpha.unsqueeze(-1) * o_i + beta.unsqueeze(-1) * P_ij_v
                ) / l_new.unsqueeze(-1)

                m_i = m_new
                l_i = l_new

            # 将当前 chunk 的结果写入输出
            O[..., i:i_end, :] = o_i

            # 更新 L (log-sum-exp)
            # 需要将 m_i 和 l_i 重塑为 (batch_size, chunk_i)
            l_i_flat = l_i.reshape(batch_size, -1)
            m_i_flat = m_i.reshape(batch_size, -1)
            L[:, i:i_end] = torch.log(l_i_flat) + m_i_flat

        # 保存所有需要的张量
        ctx.save_for_backward(Q, K, V, O, L)
        ctx.scale = scale
        ctx.is_causal = is_causal

        return O

    @staticmethod
    def backward(ctx, grad_output):
        # 暂时返回零梯度，先确保前向传播正确
        Q, K, V, O, L = ctx.saved_tensors
        grad_Q = torch.zeros_like(Q)
        grad_K = torch.zeros_like(K)
        grad_V = torch.zeros_like(V)
        return grad_Q, grad_K, grad_V, None
```
